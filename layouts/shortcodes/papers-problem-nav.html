{{- /* Previous/Next navigation between problems in the same round. Usage: {{< papers-problem-nav olympiad="ibo" year="2022" round="theoretical-1" problem="q5" >}} At the first/last problem, the missing side links back to the round overview instead. */ -}} {{- $olId := .Get "olympiad" -}} {{- $yearId := .Get "year" -}} {{- $roundId := .Get "round" -}} {{- $problemId := .Get "problem" -}} {{- $ol := index hugo.Data.papers $olId -}} {{- $yearData := index $ol $yearId -}} {{- $round := dict -}} {{- range $yearData.rounds }}{{ if eq .id $roundId }}{{ $round = . }}{{ end }}{{ end -}} {{- $problems := $round.problems -}} {{- $index := -1 -}} {{- range $i, $p := $problems }}{{ if eq $p.id $problemId }}{{ $index = $i }}{{ end }}{{ end -}} {{- $roundURL := $.Page.Parent.RelPermalink -}} {{- $prev := "" -}} {{- $next := "" -}} {{- if gt $index 0 }}{{ $prev = index $problems (sub $index 1) }}{{ end -}} {{- if lt $index (sub (len $problems) 1) }}{{ $next = index $problems (add $index 1) }}{{ end -}}
{{- if $prev }} ← Previous: {{ $prev.number }} {{- else }} ← Back to {{ $round.name }} {{- end }} {{- if $next }} Next: {{ $next.number }} → {{- else }} Back to {{ $round.name }} → {{- end }}
